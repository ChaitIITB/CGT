// Zero game: {0|0} = * (fuzzy, both sides move to same value)
G0 := {0|0};

// Positive integers: {n-1|} (Left can force move to n-1, Right cannot move)
G_pos_1 := {0|};     // 1
G_pos_1.CanonicalForm;
G_pos_1.Temperature;

G_pos_2 := {{0|}|};  // 2

G_neg_2 := {|{|0}};  // -2

// Switches: {n|-m} where n ≠ m (both sides have incentive to move)

// ============================================================
// DEPTH 2: INFINITESIMALS (atomic weights)
// ============================================================

// Infinitesimal up (↑): Left prefers moving, Right indifferent at 0-temperature
// {0|*} means: Left moves to 0, Right moves to fuzzy game
G_up := {0|{0|0}};
G_up.CanonicalForm;       // ↑
G_up.Temperature;          // 0
G_up.AtomicWeight;         // ↑

// Infinitesimal down (↓): Right prefers moving, Left indifferent
// {*|0} means: Left moves to fuzzy, Right moves to 0
G_down := {{0|0}|0};
G_down.CanonicalForm;      // ↓
G_down.Temperature;         // 0
G_down.AtomicWeight;        // ↓

// Higher order: {0|↑}
