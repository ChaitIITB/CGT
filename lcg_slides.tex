% ==================== ABSTRACT ====================
\begin{frame}{Abstract}
  We introduce \textbf{Leaf Convergence Games (LCG)}, partizan games on finite rooted DAGs where \textbf{every maximal path terminates at a designated leaf}.

  \vspace{8pt}
  Left moves on blue (\Lopt{L}) edges, Right on red (\Ropt{R}). Normal play.

  \vspace{10pt}
  \textbf{LCG provides a clean graph-theoretic lens} on classical CGT values (numbers, switches, infinitesimals, dyadics) while guaranteeing termination without surreal numbers or loopy machinery.
\end{frame}

% ==================== WHY LCG MATTERS ====================
\begin{frame}{Why LCG?}
  \textbf{Standard CGT} studies abstract game values.  
  \textbf{LCG} gives a \textbf{canonical structural model}.

  \vspace{12pt}
  \textbf{Key Contributions:}
  \begin{itemize}
    \item Guaranteed termination.
    \item Values \textbf{emerge directly from graph topology}.
    \item Bridges graph theory and CGT.
    \item Every short partizan game has an LCG representation (Representation Theorem).
  \end{itemize}

  \vspace{10pt}
  \centering
  \textbf{LCG = Where CGT values meet concrete graph structure.}
\end{frame}

% ==================== VALUE CLASSIFICATION ====================
\begin{frame}{Value Classification in LCG}
  \begin{theorem}[Numbers]
    A position is a number $\iff$ every Left option $<$ every Right option.
  \end{theorem}

  \begin{theorem}[Integers]
    Integers correspond exactly to pure directed chains (unbranched Left/Right paths).
  \end{theorem}

  \begin{theorem}[Switches]
    A position is a switch $\iff$ $\exists$ Left option $>0$ and $\exists$ Right option $<0$.
  \end{theorem}

  \begin{theorem}[Infinitesimals]
    Infinitesimal $\iff$ all Left options $\le 0$, all Right options $\ge 0$, but not all zero.
  \end{theorem}

  \vspace{8pt}
  These characterisations link \textbf{graph topology} directly to classical CGT value types.
\end{frame}

% ==================== BRANCHING CLARIFICATION ====================
\begin{frame}{Important Clarification: Branching $\neq$ Magnitude}
  \textbf{Common misconception:} More Left edges $\implies$ bigger positive number.

  \vspace{10pt}
  \textbf{Fact:} Value depends on the \textbf{values of options}, not their count.

  \begin{itemize}
    \item Duplicate identical options do not change the game value.
    \item Branching creates choice, not raw strength.
    \item True magnitude comes from depth and asymmetry in subtrees.
  \end{itemize}

  \vspace{10pt}
  \centering
  \Large Graph topology $\rightsquigarrow$ strategic options, not raw count.
\end{frame}

% ==================== HOW TO READ A GRAPH ====================
\begin{frame}{How to Read an LCG Graph — Visual Intuition}
  \begin{itemize}
    \item \textbf{Long Left-only path} $\to$ large positive integer
    \item \textbf{Long Right-only path} $\to$ large negative integer
    \item \textbf{Balanced symmetric split} $\to$ $*$
    \item \textbf{Slight Left advantage at depth} $\to$ positive infinitesimal ($\uparrow$)
    \item \textbf{Competing advantages} $\to$ switch ($\pm n$)
  \end{itemize}

  \vspace{15pt}
  \centering
  \textbf{LCG makes CGT values visually readable from the graph structure.}
\end{frame}


\begin{frame}{Graph Structure of Numbers}
\textbf{How numbers arise in LCG graphs}

\vspace{5pt}

\begin{itemize}
  \item \textbf{Integers:} Pure chains
  \begin{itemize}
    \item $1 = \{0 \mid\}$ (single Left step)
    \item $2 = \{1 \mid\}$ (two-step chain)
  \end{itemize}

  \vspace{5pt}

  \item \textbf{Fractions:} Ordered splits
  \begin{itemize}
    \item $\frac{1}{2} = \{0 \mid 1\}$
    \item $\frac{3}{2} = \{1 \mid 2\}$
  \end{itemize}

  \vspace{5pt}

  \item \textbf{Key Rule:}
  \begin{itemize}
    \item Left subtree gives \emph{lower values}
    \item Right subtree gives \emph{higher values}
  \end{itemize}
\end{itemize}

\vspace{10pt}
\textbf{Insight:} Numbers arise from \emph{ordered asymmetry}, not branching.
\end{frame}

% ==================== FORMAL FRAMEWORK ====================
\begin{frame}{Formal Framework}
  \textbf{LCG Position:} Finite rooted DAG + terminal leaves + L/R edge labels + convergence property.

  \vspace{12pt}
  \textbf{Value Definition:}
  \[
  \Gv{v} = \bigl\{ \Gv{c} : c \in \mathcal{L}(v) \;\big|\; \Gv{c} : c \in \mathcal{R}(v) \bigr\}
  \]
  with $\Gv{\tau} = 0$ for terminals.
\end{frame}

\begin{frame}{Numbers in Leaf Convergence Games}
\textbf{Theorem (Number Characterisation).}

A position $v$ in an LCG is a \textbf{number} iff
\[
\max \mathcal{G}(\mathcal{L}(v)) \;<\; \min \mathcal{G}(\mathcal{R}(v)).
\]

\vspace{10pt}

\textbf{Interpretation:}
\begin{itemize}
  \item Every Left option is strictly \emph{worse} than every Right option.
  \item There is a \textbf{gap} between Left and Right outcomes.
  \item The position equals the \textbf{simplest number in that gap}.
\end{itemize}

\vspace{10pt}

\textbf{Consequences:}
\begin{itemize}
  \item Values are dyadic rationals ($\frac{m}{2^k}$).
  \item No “first-player advantage” ambiguity.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Computational Analysis}
  \textbf{Python evaluator:}
  \begin{lstlisting}
def evaluate_lcg(nodes, edges, root, terminals):
    memo = {}
    def eval_node(v):
        if v in terminals: return ([], [])
        left_opts = [eval_node(c) for s,c,col in edges if s==v and col=='L']
        right_opts = [eval_node(c) for s,c,col in edges if s==v and col=='R']
        return (left_opts, right_opts)
    return eval_node(root)
  \end{lstlisting}
\end{frame}
